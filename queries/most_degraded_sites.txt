--Query
select RED, ESTACION, SUM(EFI_VOZ)EFI_VOZ, SUM(EFI_DAT)EFI_DAT, SUM(RTWP)RTWP, SUM(RSSR)RSSR, SUM(CCR)CCR, SUM(CSFR)CSFR, SUM(ERL)ERL, SUM(GB)GB
FROM(
select RED, ESTACION, avg((-1*(CASE WHEN EFI_VOZ <> 0 THEN EFI_VOZ ELSE NULL END)))EFI_VOZ, avg((-1*(CASE WHEN EFI_DAT <> 0 THEN EFI_DAT ELSE NULL END)))EFI_DAT, -avg((-1*rtwp)/100)RTWP, avg((-1*(CASE WHEN RSSR <> 0 THEN RSSR ELSE NULL END)))RSSR, avg((-1*(CASE WHEN CCR <> 0 THEN CCR ELSE NULL END)))CCR, avg((-1*(CASE WHEN CSFR <> 0 THEN CSFR ELSE NULL END)))CSFR, avg((-1*(CASE WHEN GB <> 0 THEN GB ELSE NULL END)))GB, avg((-1*(CASE WHEN ERL <> 0 THEN ERL ELSE NULL END)))ERL
from KPI_EEBB.HTT_KPIS_EEBB
WHERE FECHA >= SYSDATE - 30
AND FECHA < SYSDATE - 1
group by RED, ESTACION
UNION ALL
select RED, ESTACION, avg(((CASE WHEN EFI_VOZ <> 0 THEN EFI_VOZ ELSE NULL END)))EFI_VOZ, avg(((CASE WHEN EFI_DAT <> 0 THEN EFI_DAT ELSE NULL END)))EFI_DAT, -avg((rtwp)/100)RTWP, avg(((CASE WHEN RSSR <> 0 THEN RSSR ELSE NULL END)))RSSR, avg(((CASE WHEN CCR <> 0 THEN CCR ELSE NULL END)))CCR, -avg(((CASE WHEN CSFR <> 0 THEN CSFR ELSE NULL END)))CSFR, avg(((CASE WHEN GB <> 0 THEN GB ELSE NULL END)))GB, avg(((CASE WHEN ERL <> 0 THEN ERL ELSE NULL END)))ERL
from KPI_EEBB.HTT_KPIS_EEBB
WHERE FECHA >= SYSDATE - 1
group by RED, ESTACION
)
group by RED, ESTACION
